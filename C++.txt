if+else
#include <iostream>
#include <windows.h>

int main() {
	SetConsoleOutputCP(CP_UTF8);

	int num;
	std::cout << "Секретная кодировка, в случае неправильного кода вы будете заблокированы: ";
	std::cin >> num;

	bool is_has_car = true;
	if (num > 26 && is_has_car == true) {                               // &&-и,||-или
		std::cout << "Секретная кодировка успешно пройдена!";
		if (num == 28) {
			std::cout << "Привет от fleerq0";
		}
	}
	else if (num == 2) {
		std::cout << "не правильно";
	}
	else {
		std::cout << "Братишка,пизда тебе";
	}
	
	
	
	return 0;
}
